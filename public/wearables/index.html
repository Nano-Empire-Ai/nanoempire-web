HOW TO LOAD ON META RAY-BAN DISPLAY GLASSES + ANDROID DAT v1.0.1

1. Direct Glasses WebApp Launch · 2. Native Android DAT SDK (`com.meta.wearable:mwdat-display:1.0.1`)

WebApp Launch on Glasses: Open the Meta AI phone app → App Settings → Apps → Web Apps → Connect Web App → Enter https://www.nanoempireai.com/wearables/. The glasses automatically fetch /wearables/.well-known/meta-wearables-manifest.json and monochrome icon artwork. 

// Android DAT v1.0.1 (com.meta.wearable:mwdat-core:1.0.1 + mwdat-display:1.0.1)
val selector = AutoDeviceSelector(filter = { device -> device.isDisplayCapable() })
Wearables.createSession(selector).fold(
    onSuccess = { session ->
        lifecycleScope.launch {
            session.state.collect { state ->
                if (state == DeviceSessionState.STARTED) {
                    session.addDisplay().onSuccess { display ->
                        display.sendContent {
                            flexBox(gap = 10, padding = 20, background = FlexBoxBackground.CARD) {
                                text("RECALLGUARD HAZARD MATCH", style = TextStyle.HEADING)
                                text("Fisher-Price Rock 'n Play (CPSC)", style = TextStyle.BODY)
                                buttonGroup {
                                    button(
                                        label = "Quarantine SKU",
                                        style = ButtonStyle.PRIMARY,
                                        actionRole = ActionRole.PRIMARY,
                                        iconName = IconName.CHECKMARK,
                                        onClick = { acknowledgeQuarantine() }
                                    )
                                }
                            }
                        }
                    }
                }
            }
        }
        session.start() // Fire-and-forget per DAT 1.0.1 convention
    },
    onFailure = { error, _ -> showError(error.description) }
)
